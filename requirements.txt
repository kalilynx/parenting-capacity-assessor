"""Local RAG knowledge base over user-uploaded files."""
from __future__ import annotations

from pathlib import Path
from typing import List

from langchain_chroma import Chroma
from langchain_community.document_loaders import Docx2txtLoader, PyPDFLoader, TextLoader
from langchain_community.embeddings import HuggingFaceEmbeddings
from langchain_text_splitters import RecursiveCharacterTextSplitter

from config import (
    CHUNK_OVERLAP,
    CHUNK_SIZE,
    EMBEDDING_MODEL,
    KNOWLEDGE_DIR,
    RETRIEVAL_TOP_K,
)


class KnowledgeBase:
    """Thin wrapper around a persistent ChromaDB collection."""

    def __init__(self, collection_name: str = "pca_knowledge"):
        self.persist_dir = KNOWLEDGE_DIR / "_chroma"
        self.persist_dir.mkdir(parents=True, exist_ok=True)

        self.embeddings = HuggingFaceEmbeddings(
            model_name=EMBEDDING_MODEL,
            model_kwargs={"device": "cpu"},
            encode_kwargs={"normalize_embeddings": True},
        )

        self.vectorstore = Chroma(
            collection_name=collection_name,
            embedding_function=self.embeddings,
            persist_directory=str(self.persist_dir),
        )
        self.splitter = RecursiveCharacterTextSplitter(
            chunk_size=CHUNK_SIZE,
            chunk_overlap=CHUNK_OVERLAP,
            separators=["\n\n", "\n", ". ", " ", ""],
        )

    def add_file(self, path: Path) -> int:
        """Ingest a single file. Returns number of chunks stored."""
        loader = self._loader_for(path)
        docs = loader.load()
        for d in docs:
            d.metadata["source"] = path.name
        chunks = self.splitter.split_documents(docs)
        if not chunks:
            return 0
        self.vectorstore.add_documents(chunks)
        self.vectorstore.persist()
        return len(chunks)

    def add_skill_doc(self, text: str, name: str = "skill_doc.txt") -> int:
        tmp = KNOWLEDGE_DIR / name
        tmp.write_text(text, encoding="utf-8")
        return self.add_file(tmp)

    def list_sources(self) -> List[str]:
        if not self.vectorstore.get()["ids"]:
            return []
        all_meta = self.vectorstore.get(include=["metadatas"])["metadatas"]
        return sorted({m.get("source", "?") for m in all_meta})

    def count(self) -> int:
        return len(self.vectorstore.get()["ids"])

    def clear(self) -> None:
        self.vectorstore.delete_collection()
        self.vectorstore = Chroma(
            collection_name="pca_knowledge",
            embedding_function=self.embeddings,
            persist_directory=str(self.persist_dir),
        )

    def retrieve(self, query: str, k: int = RETRIEVAL_TOP_K) -> str:
        if self.count() == 0:
            return "(knowledge base empty — no reference files uploaded)"
        docs = self.vectorstore.similarity_search(query, k=k)
        parts = []
        for i, d in enumerate(docs, 1):
            src = d.metadata.get("source", "unknown")
            parts.append(f"[{i}] (source: {src})\n{d.page_content.strip()}")
        return "\n\n---\n\n".join(parts)

    @staticmethod
    def _loader_for(path: Path):
        suffix = path.suffix.lower()
        if suffix == ".pdf":
            return PyPDFLoader(str(path))
        if suffix in (".docx", ".doc"):
            return Docx2txtLoader(str(path))
        if suffix in (".txt", ".md", ".markdown"):
            return TextLoader(str(path), encoding="utf-8")
        return TextLoader(str(path), encoding="utf-8")
